# Qt stylesheets need an individual SVG URL for indicator images. Generate the
# glyph from the shared sprite, so the shape is still owned by one source.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../VERSION" "${STACKJUNKIE_SHARED}/icons.svg" "${STACKJUNKIE_SHARED}/tokens.json")

function(stackjunkie_sprite_svg NAME OUTPUT)
  string(FIND "${STACKJUNKIE_ICON_SPRITE}" "<symbol id=\"${NAME}\"" SYMBOL_START)
  if(SYMBOL_START LESS 0)
    message(FATAL_ERROR "Shared icon sprite must contain the ${NAME} symbol")
  endif()
  string(SUBSTRING "${STACKJUNKIE_ICON_SPRITE}" ${SYMBOL_START} -1 SYMBOL_TAIL)
  string(FIND "${SYMBOL_TAIL}" "</symbol>" SYMBOL_END)
  math(EXPR SYMBOL_LENGTH "${SYMBOL_END} + 9")
  string(SUBSTRING "${SYMBOL_TAIL}" 0 ${SYMBOL_LENGTH} SVG)
  string(REPLACE "<symbol " "<svg xmlns=\"http://www.w3.org/2000/svg\" " SVG "${SVG}")
  string(REPLACE "</symbol>" "</svg>" SVG "${SVG}")
  set(${OUTPUT} "${SVG}" PARENT_SCOPE)
endfunction()
stackjunkie_sprite_svg(check STACKJUNKIE_CHECK_SVG)
stackjunkie_sprite_svg(chevron-down STACKJUNKIE_CHEVRON_SVG)
set(STACKJUNKIE_INDICATOR_RESOURCES "")
foreach(DIRECTION IN ITEMS clean)
  foreach(MODE IN ITEMS light dark)
    string(JSON GLYPH_COLOR GET "${STACKJUNKIE_TOKEN_JSON}" directions ${DIRECTION} ${MODE} onAccent)
    string(REPLACE "currentColor" "${GLYPH_COLOR}" CHECK_GLYPH "${STACKJUNKIE_CHECK_SVG}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/check-${DIRECTION}-${MODE}.svg" "${CHECK_GLYPH}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mixed-${DIRECTION}-${MODE}.svg" "<svg xmlns=\"http://www.w3.org/2000/svg\" viewBox=\"0 0 24 24\"><path d=\"M6 12h12\" stroke=\"${GLYPH_COLOR}\" stroke-width=\"3\" stroke-linecap=\"round\"/></svg>")
    string(JSON ARROW_COLOR GET "${STACKJUNKIE_TOKEN_JSON}" directions ${DIRECTION} ${MODE} text)
    string(REPLACE "currentColor" "${ARROW_COLOR}" DOWN_GLYPH "${STACKJUNKIE_CHEVRON_SVG}")
    string(REGEX REPLACE "^(<svg[^>]*>)" "\\1<g transform=\"rotate(180 12 12)\">" UP_GLYPH "${DOWN_GLYPH}")
    string(REPLACE "</svg>" "</g></svg>" UP_GLYPH "${UP_GLYPH}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/chevron-down-${DIRECTION}-${MODE}.svg" "${DOWN_GLYPH}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/chevron-up-${DIRECTION}-${MODE}.svg" "${UP_GLYPH}")
    foreach(GLYPH IN ITEMS check mixed chevron-down chevron-up)
      string(APPEND STACKJUNKIE_INDICATOR_RESOURCES "  <file alias=\"${GLYPH}-${DIRECTION}-${MODE}.svg\">${CMAKE_CURRENT_BINARY_DIR}/${GLYPH}-${DIRECTION}-${MODE}.svg</file>\n")
    endforeach()
  endforeach()
endforeach()

# Release builds of consumers refuse to compile while any STACKJUNKIE_STUB remains, even without NDEBUG.
# Compare $<CONFIG> by name: $<CONFIG:...> on an imported target also matches its mapped configuration,
# which would block Stubs in a Debug app linking a Release-only package.
target_compile_definitions(stackjunkie_widgets INTERFACE
  $<$<OR:$<STREQUAL:$<CONFIG>,Release>,$<STREQUAL:$<CONFIG>,RelWithDebInfo>,$<STREQUAL:$<CONFIG>,MinSizeRel>>:STACKJUNKIE_RELEASE>)
if(NOT BUILD_SHARED_LIBS)
  target_compile_definitions(stackjunkie_widgets PUBLIC STACKJUNKIE_WIDGETS_STATIC_DEFINE)
endif()

endif()
